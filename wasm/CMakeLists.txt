cmake_minimum_required(VERSION 3.21)
project(MeshKitWasm LANGUAGES C CXX)
if(NOT EMSCRIPTEN)
    message(FATAL_ERROR "Configure with emcmake cmake -S wasm -B build/wasm")
endif()
set(MESHKIT_BUILD_QT OFF)
set(MESHKIT_BUILD_CLI OFF)
set(MESHKIT_BUILD_TESTS OFF)
add_compile_options(-fexceptions)
add_subdirectory(.. meshkit-core EXCLUDE_FROM_ALL)
add_executable(meshkit_browser bindings.cpp)
target_link_libraries(meshkit_browser PRIVATE meshkit::core)
target_link_options(meshkit_browser PRIVATE --bind -fexceptions --no-entry
    "-sMODULARIZE=1" "-sEXPORT_ES6=1" "-sALLOW_MEMORY_GROWTH=1" "-sENVIRONMENT=web,worker,node")
set_target_properties(meshkit_browser PROPERTIES OUTPUT_NAME meshkit SUFFIX ".mjs")
